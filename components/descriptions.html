Descriptions 组件 ​

列表形式展示多个字段

示例 ​

vue
<template>
  <c-descriptions v-bind="{ schemas: descSchemas, data }"></c-descriptions>
</template>
<script lang="ts" setup>
import { SchemasType } from '@/components/c-descriptions/type';

const data = ref({});

const descSchemas: SchemasType[] = [
  { label: '门店编码', field: 'custCode' },
  { label: '门店类型', field: 'storeType', render: matchType, component: 'Tag' },
  { label: '门店性质', field: 'storeLevel', render: matchLevel, component: 'Tag' },
  { label: '门店分级', field: 'hisBranch', render: matchBranch },
  {
    label: '上级客户',
    field: 'parentCode',
    render: () =>
      unref(storeParent)
        ?.map(({ custName }) => custName)
        .join('，'),
  },
  {
    label: '所在省市',
    field: 'city',
    render: (_v, { provinceName = '', cityName = '' }) => provinceName && `${provinceName} / ${cityName}`,
  },
  { label: '详细地址', field: 'address' },
  { label: '联系电话', field: 'tel' },
  { label: '备注', field: 'remark' },
];
</script>

Props ​

参数名类型默认值说明
dataObject{}数据对象
schemasArray<SchemasType>-配置项数组
titleString-标题文本
borderedBooleanfalse是否显示边框
colonBooleantrue是否显示标签后的冒号
contentStyleCSSProperties-内容区域样式（CSS 属性对象）
labelWidthString-标签宽度（如：'120px'）
labelStyleCSSProperties{}标签样式（CSS 属性对象）
layoutString'horizontal'布局方式，可选：'horizontal'/ 'vertical'
sizeString'default'尺寸，可选：'default' / 'small'
classString-自定义类名
SchemasType
ts
export interface SchemasType {
  label?: string;
  field?: string;
  render?: (val: any, data: Recordable) => string | number | Object | Array<any>; //渲染
  ifShow?: boolean | ((val: any, data: Recordable) => boolean);
  component?: ComponentType;
  componentProps?: Recordable | ((val: any, data: Recordable) => Recordable);
  slot?: string;
  layout?: 'horizontal' | 'vertical';
  class?: string;
  labelWidth?: string;
}
ComponentType
ts
type ComponentType = 'Image' | 'List' | 'Tag' | 'Html' | 'Card' | 'File';
