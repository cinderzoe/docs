ButtonPicker 按钮选择器 ​

示例 ​

vue
<template>
  <ButtonPicker v-bind="buttonPickerProps" @confirm="toCreatePage" v-if="hasPermission(PermissionEnum.ReportAdd)" />
</template>
<script setup lang="ts">
import ButtonPicker from '@/components/ButtonPicker.vue';
const buttonPickerProps = {
  text: '写报告',
  icon: 'edit-pen',
  api: reportTypeApi,
  labelField: 'dictLabel',
  valueField: 'dictValue',
};

function toCreatePage({ value }) {
  uni.navigateTo({ url: `/pages/report/create?reportType=${value}` });
  uni.$once('reloadList', reset);
}
</script>

Props ​

属性名类型默认值说明
iconString'plus'图标类型
iconSizeNumber30图标大小
textString-文本内容
optionsArray<FormOptionsType[]>-选项数组
apiFunctionnull异步请求函数
labelFieldString'label'标签字段名
valueFieldString'value'值字段名
resultFieldString-结果字段名
resultParamsArray<string>-结果参数数组
paramsObject{}参数对象
bottomHeightNumber70底部高度
